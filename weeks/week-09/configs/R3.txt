! ==========================================================
!  R3 - OSPF domain (loopbacks + tag practice) [AS-BUILT]
!  e0/0 = to R1, 10.1.3.0/24 (OSPF 100, area 0)
!  Lo1-Lo7 = 30.30.x.3/32 (OSPF area 0)
!  Lo200-204,208 = various subnets (prefix-list practice)
!  Lo205-207 = 192.168.x.x (tag practice, redistributed connected)
!  BUGS FIXED: duplicate Lo203/Lo204, missing "ip ospf 100 area 0"
!              on Lo205-207, "match ip add" -> "match ip address"
! ==========================================================
hostname R3
!
interface Ethernet0/0
 ip address 10.1.3.3 255.255.255.0
 ip ospf 100 area 0
 no shutdown
!

! ============ Route tagging ============
ip prefix-list Tag100 seq 100 permit 192.168.1.0/24
ip prefix-list Tag100 seq 101 permit 192.168.2.0/24
!
ip prefix-list Tag200 seq 100 permit 192.168.3.0/24
ip prefix-list Tag200 seq 101 permit 192.168.4.0/24
!
route-map ospf-tag permit 100
 match ip address prefix-list Tag100
 set tag 100
route-map ospf-tag permit 101
 match ip address prefix-list Tag200
 set tag 200
route-map ospf-tag permit 110
 ! catch-all: unmatched connected routes get no tag
!
router ospf 100
 redistribute connected subnets route-map ospf-tag
!
end
