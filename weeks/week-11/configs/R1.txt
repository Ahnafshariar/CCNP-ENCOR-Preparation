! ==========================================================
!  R1 - iBGP client (AS 100)  [AS-BUILT, final v4]
!  Lo100/101 NOT in OSPF — BGP owns these exclusively
!  network statements use /32 to match connected route exactly
! ==========================================================
hostname R1
!
interface Loopback0
 ip address 1.1.1.1 255.255.255.255
 ip ospf 100 area 0
interface Loopback100
 ip address 192.168.1.1 255.255.255.255
interface Loopback101
 ip address 192.168.11.1 255.255.255.255
!
interface Ethernet0/0
 ip address 10.1.4.1 255.255.255.0
 ip ospf 100 area 0
 no shutdown
interface Ethernet0/1
 ip address 10.1.5.1 255.255.255.0
 ip ospf 100 area 0
 no shutdown
!
router bgp 100
 neighbor PG3 peer-group
 neighbor PG3 remote-as 100
 neighbor PG3 update-source Loopback0
 neighbor 4.4.4.4 peer-group PG3
 neighbor 5.5.5.5 peer-group PG3
 network 192.168.1.1 mask 255.255.255.255
 network 192.168.11.1 mask 255.255.255.255
!
end
